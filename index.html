async function dispatchBackgroundTaskWithFallback(requestPayload) { const bgCandidates = Array.from(new Set([ GDeepSystemConfig.Models.ANTIGRAVITY?.BACKGROUND_MODEL, GDeepSystemConfig.Models.BACKGROUND.FAST, GDeepSystemConfig.Models.BACKGROUND.LITE, 'gemini-2.5-flash-lite', 'gemini-3.6-flash' ].filter(Boolean))); const pool = (apiKeyPool.value && apiKeyPool.value.length > 0) ? apiKeyPool.value : []; if (!pool.length) return null; const gateway = 'v1beta'; const isLongTask = Number(requestPayload?.generationConfig?.maxOutputTokens) > 8192; const isJsonMode = requestPayload?.generationConfig?.responseMimeType === 'application/json'; const sysInstr = requestPayload?.systemInstruction?.parts?.map(p => p?.text || '').join('\n').trim() || ''; const timeoutMs = isLongTask ? 60000 : 25000; for (const modelId of bgCandidates) { const candidateKeys = getKeysForModel(modelId, { allowCoolingFallback: false }) || []; if (!candidateKeys.length) continue; const maxAttempts = Math.min(4, candidateKeys.length); let stripJsonMime = false; let stripGoogleSearch = false; for (let attempt = 0; attempt < maxAttempts; attempt++) { const healthyKeyObj = candidateKeys[attempt]; if (!healthyKeyObj) continue; const reqCtrl = new AbortController(); const reqTimer = setTimeout(() => reqCtrl.abort(), timeoutMs); const startTs = performance.now(); try { // LỚP CHÍNH: Google Antigravity Interactions API (dùng stream cho tác vụ dài trên Mobile để tránh socket idle timeout) if (typeof GDeepAntigravityCore !== 'undefined') { try { const inputMeta = GDeepAntigravityCore.buildInteractionsInput(requestPayload?.contents || [], { preferStateful: false }); let collectedText = ''; await GDeepAntigravityCore.runInteraction( 'model', { modelId, systemInstruction: sysInstr, input: inputMeta.input, taskType: isLongTask ? 'agent' : 'background', jsonMode: isJsonMode, stripJsonMime, stripGoogleSearch, maxOutputTokens: Number(requestPayload?.generationConfig?.maxOutputTokens) || 1024, temperature: requestPayload?.generationConfig?.temperature ?? null, stream: isLongTask && GDeepAntigravityCore.isStreamingSupportedInBrowser() }, healthyKeyObj.key, reqCtrl.signal, (chunk) => { const cParts = chunk?.candidates?.[0]?.content?.parts || []; for (const cp of cParts) { if (!cp.thought && typeof cp.text === 'string') collectedText += cp.text; } } ); if (collectedText.trim()) { const latencyMs = Math.round(performance.now() - startTs); markKeyModelSuccess(healthyKeyObj, modelId, { latencyMs }); return { candidates: [{ content: { parts: [{ text: collectedText }] } }] }; } } catch (interErr) { if (reqCtrl.signal.aborted) throw interErr; const iMsg = String(interErr?.message || ''); const iKind = GeminiPayloadAdapter.classifyApiError(interErr?.httpStatus || 0, iMsg); if (iKind === 'feature_tier_restricted' && !stripGoogleSearch) { stripGoogleSearch = true; attempt--; continue; } if (iKind === 'payload_error' && !stripJsonMime && isJsonMode) { stripJsonMime = true; attempt--; continue; } if (iKind === 'invalid_key' || iKind === 'quota_rpm' || iKind === 'quota_rpd') { markKeyModelFailure(healthyKeyObj, modelId, interErr.httpStatus, iMsg, interErr.retryAfterSec || null); if (interErr.httpStatus === 429) await new Promise(r => setTimeout(r, 350)); continue; } // Các trường hợp model cũ chưa bật Interactions API -> rơi mượt xuống :generateContent bên dưới } } // LỚP DỰ PHÒNG: Legacy :generateContent const compliantPayload = GeminiPayloadAdapter.buildPayload(modelId, requestPayload, { taskType: isLongTask ? 'agent' : 'background', stripJsonMime, stripGoogleSearch }); const res = await fetch(`https://generativelanguage.googleapis.com/${gateway}/models/${modelId}:generateContent?key=${healthyKeyObj.key}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(compliantPayload), signal: reqCtrl.signal }); if (res.ok) { const latencyMs = Math.round(performance.now() - startTs); markKeyModelSuccess(healthyKeyObj, modelId, { latencyMs }); return await res.json(); } const errData = await res.json().catch(() => ({})); const errMsg = String(errData?.error?.message || ''); const errKind = GeminiPayloadAdapter.classifyApiError(res.status, errMsg); const retryAfterSec = Number(res.headers.get('Retry-After')) || null; if (errKind === 'feature_tier_restricted' && !stripGoogleSearch) { stripGoogleSearch = true; attempt--; continue; } if (errKind === 'payload_error' && !stripJsonMime && compliantPayload.generationConfig?.responseMimeType) { stripJsonMime = true; attempt--; continue; } markKeyModelFailure(healthyKeyObj, modelId, res.status, errMsg, retryAfterSec); if (res.status === 429 || res.status >= 500) { await new Promise(r => setTimeout(r, 350)); } } catch (netErr) { markKeyModelFailure(healthyKeyObj, modelId, null, netErr?.message || 'network'); await new Promise(r => setTimeout(r, 350)); } finally { clearTimeout(reqTimer); } } } return null; }
